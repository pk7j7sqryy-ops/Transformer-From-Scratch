## 一、Tokenizer（分词器）

### 是什么
把文本转成数字 ID（token ID），让模型能处理。翻译项目用了两个分词器：
- `bert-base-chinese`：中文分词器，词表 21128
- `bert-base-uncased`：英文分词器，词表 30522

### 为什么用
模型不认识文字，只认识数字。Tokenizer 是文本和数字之间的桥梁。

### 代码演示
```python
from transformers import AutoTokenizer

# 加载分词器
src_tokenizer = AutoTokenizer.from_pretrained('google-bert/bert-base-chinese')
tgt_tokenizer = AutoTokenizer.from_pretrained('google-bert/bert-base-uncased')

# encode: 文本 → token IDs
src_ids = src_tokenizer.encode("你好世界")
print(src_ids)  # [101, 872, 1962, 102]

tgt_ids = tgt_tokenizer.encode("hello world")
print(tgt_ids)  # [101, 3403, 102]

# decode: token IDs → 文本（去掉特殊 token）
text = tgt_tokenizer.decode([101, 3403, 102], skip_special_tokens=True)
print(text)  # "hello world"

# 不去掉特殊 token
text_full = tgt_tokenizer.decode([101, 3403, 102], skip_special_tokens=False)
print(text_full)  # "[CLS] hello world [SEP]"

# 特殊 token ID
print(src_tokenizer.cls_token_id)  # 101 ([CLS] 句首)
print(src_tokenizer.sep_token_id)  # 102 ([SEP] 句尾)
print(src_tokenizer.pad_token_id)  # 0   ([PAD] 填充)
```

```python
from torch.utils.data import DataLoader
from torch.nn.utils.rnn import pad_sequence

def collate_batch(batch, pad_id=0):
    """把不等长的句子补 PAD 到等长"""
    src_ids = [torch.tensor(item['src_ids']) for item in batch]
    tgt_ids = [torch.tensor(item['tgt_ids']) for item in batch]
    src = pad_sequence(src_ids, batch_first=True, padding_value=pad_id)
    tgt = pad_sequence(tgt_ids, batch_first=True, padding_value=pad_id)
    return {'src': src, 'tgt': tgt}

# 示例：batch 内 4 条句子，不等长

# [101, 639, 2828, 102, 0]    → [101, 639, 2828, 102, 0, 0]  ← 补 0
# [101, 791, 1921, 6843, 102] → [101, 791, 1921, 6843, 102]    ← 最长，不补
# [101, 800, 1962, 102]        → [101, 800, 1962, 102, 0, 0]  ← 补 0
```

## 三、Embedding 与位置编码

### Embedding 是什么
查表操作，把 token ID 变成向量。`nn.Embedding(vocab_size, d_model)` 就是一个大矩阵，每行对应一个词的向量。

### 代码演示
```python
import torch
import torch.nn as nn
import math

# Embedding 层
vocab_size = 30522
d_model = 512
embedding = nn.Embedding(vocab_size, d_model)

# token ID → 向量
token_ids = torch.tensor([[101, 3403, 102]])  # [1, 3]
emb = embedding(token_ids)  # [1, 3, 512]
print(emb.shape)
print(f"不乘 √d_model 范数: {emb.norm(dim=-1)}")  # ≈ √512 ≈ 22.6

# 乘 √d_model
emb_scaled = emb * math.sqrt(d_model)
print(f"乘 √d_model 范数: {emb_scaled.norm(dim=-1)}")  # ≈ 512

# 位置编码
pe = PositionalEncoding(d_model, max_len=5000)
result = pe(emb_scaled)  # embedding + 位置编码
```

### 为什么乘 √d_model
- 不乘：embedding 范数 ≈ √d_model ≈ 22，PE 范数 ≈ 16，量级接近，PE 干扰词义
- 乘了：embedding 范数 ≈ d_model = 512，PE 范数 ≈ 16，embedding 碾压 PE

### 位置编码实现
```python
class PositionalEncoding(nn.Module):
    def __init__(self, d_model, max_len=5000):
        super().__init__()
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len).unsqueeze(1).float()
        div_term = torch.exp(torch.arange(0, d_model, 2).float() *
                             -(math.log(10000.0) / d_model))
        pe[:, 0::2] = torch.sin(position * div_term)  # 偶数维用 sin
        pe[:, 1::2] = torch.cos(position * div_term)  # 奇数维用 cos
        self.register_buffer('pe', pe.unsqueeze(0))   # 不参与梯度更新

    def forward(self, x):
        return x + self.pe[:, :x.size(1)]
```

## 四、Mask（掩码）

### 为什么需要 mask
数据已经用 0 填充了 PAD，但 embedding 层不认识 0 是 PAD，会把 0 当普通 token 查表得到向量。如果不加 mask，PAD 会参与 attention 计算，污染真实 token 的表示。

### 代码演示
```python
import torch

# ============ 1. Padding Mask ============
def make_padding_mask(tokens, pad_id=0):
    """True=真实token(可见), False=PAD(屏蔽)"""
    return (tokens != pad_id).unsqueeze(1).unsqueeze(1)
    # [B, S] → [B, 1, 1, S]

tokens = torch.tensor([[101, 872, 102, 0, 0]])
mask = make_padding_mask(tokens, pad_id=0)
# [[[True, True, True, False, False]]]  shape [1, 1, 1, 5]

# ============ 2. Causal Mask（因果掩码）============
def make_causal_mask(size, device=None):
    """下三角矩阵，True=可见(过去), False=屏蔽(未来)"""
    return torch.tril(torch.ones(size, size, dtype=torch.bool, device=device))
    # [T, T]

causal = make_causal_mask(4)
# [[True,  False, False, False],
#  [True,  True,  False, False],
#  [True,  True,  True,  False],
#  [True,  True,  True,  True]]

# ============ 3. Decoder Mask ============
def make_decoder_mask(tokens, pad_id=0):
    """padding mask + causal mask 组合"""
    padding_mask = make_padding_mask(tokens, pad_id)  # [B, 1, 1, T]
    causal_mask = make_causal_mask(tokens.size(1), device=tokens.device)  # [T, T]
    return padding_mask & causal_mask  # [B, 1, T, T]
```

### 在 attention 中怎么用
```python
def attention(query, key, value, mask=None):
    d_k = query.size(-1)
    scores = (query @ key.mT) / math.sqrt(d_k)  # [B, H, L, L]
    if mask is not None:
        # PAD/未来位置 score 变成极小值 → softmax 后权重为 0
        scores = scores.masked_fill(mask == 0, torch.finfo(scores.dtype).min)
    attn = torch.softmax(scores, dim=-1)
    return attn @ value
```

## 五、Batch 与数据流

### Batch 类做了什么
```python
class Batch:
    def __init__(self, src, tgt, pad_id=0):
        self.src = src                            # [B, S] 中文 token IDs
        self.src_mask = make_padding_mask(src, pad_id)  # [B, 1, 1, S]
        self.tgt = tgt[:, :-1]    # 去掉最后一个，decoder 输入
        self.tgt_y = tgt[:, 1:]    # 去掉第一个，正确答案（标签）
        self.tgt_mask = make_decoder_mask(self.tgt, pad_id)
        self.ntokens = (self.tgt_y != pad_id).sum()  # 非 PAD token 数

# 示例
tgt = torch.tensor([[101, 3403, 102, 0, 0]])  # [CLS] hello world [PAD] [PAD]
batch = Batch(src, tgt, pad_id=0)

# batch.tgt   = [101, 3403, 102, 0, 0]  ← decoder 输入
# batch.tgt_y = [3403, 102, 0, 0, 0]    ← 正确答案
# 模型看到 BOS(101) → 预测 hello(3403)
# 模型看到 BOS+hello → 预测 world(102)
```

## 六、Loss 计算

### LabelSmoothing 损失
```python
class LabelSmoothing(nn.Module):
    def __init__(self, size, padding_idx=0, smoothing=0.1):
        super().__init__()
        self.criterion = nn.KLDivLoss(reduction='sum')
        self.padding_idx = padding_idx
        self.confidence = 1.0 - smoothing  # 0.9
        self.smoothing = smoothing          # 0.1
        self.size = size                   # 词表大小 30522

    def forward(self, x, target):
        # x: [28, 30522] log 概率
        # target: [28] 正确答案索引

        # 构造平滑标签
        true_dist = x.data.clone()
        true_dist.fill_(self.smoothing / (self.size - 2))  # 全部填 0.0000033
        true_dist.scatter_(1, target.data.unsqueeze(1), self.confidence)  # 正确词位置填 0.9
        true_dist[:, self.padding_idx] = 0  # PAD 位置清零

        # KLDivLoss 比较模型预测分布 x 和 平滑标签 true_dist
        return self.criterion(x, true_dist.clone().detach())

# 正常标签:  [0, 0, 1, 0]  → 模型拼命推到 100%，容易过拟合
# 平滑标签:  [0.025, 0.025, 0.9, 0.025] → 给其他词留概率，防止过度自信
```

### 为什么用 KLDivLoss 不用 CrossEntropyLoss
```python
# Generator 已经做了 log_softmax，输出是 log 概率
class Generator(nn.Module):
    def __init__(self, d_model, vocab_size):
        super().__init__()
        self.proj = nn.Linear(d_model, vocab_size)

    def forward(self, x):
        return torch.log_softmax(self.proj(x), dim=-1)  # ← log 概率

# CrossEntropyLoss 内部会再做一次 log_softmax，重复了
# KLDivLoss 直接比较两个概率分布，输入正好是 log 概率，匹配
```

### Loss 计算完整流程
```python
# 1. 前向传播
out = model(batch.src, batch.tgt, batch.src_mask, batch.tgt_mask)
# [4, 7, 256]  ← 4条句子，每句7个位置，256维

# 2. 映射到词表
out = model.generator(out)
# [4, 7, 30522]  ← 每个位置一个 30522 维概率分布

# 3. reshape 拍平
out_flat = out.reshape(-1, out.size(-1))   # [28, 30522]
tgt_y_flat = batch.tgt_y.reshape(-1)        # [28]

# 4. 算 loss
loss = loss_fn(out_flat, tgt_y_flat)        # 标量

# 5. 归一化（除以非 PAD token 数）
loss_normalized = loss.item() / max(1, batch.ntokens.item())
# 不归一化: batch 越大 loss 越大，无法对比
# 归一化后: 平均每 token loss，和 batch_size 无关
```

### 为什么不需要 argmax
```python
# 训练/评估：正确答案从数据里来
tgt_y = [3403, 102, 0, 0]  # ← 直接从数据取的，是已知索引
loss = loss_fn(out, tgt_y)  # ← 用索引去 out 里取概率算损失
# 不需要 argmax，argmax 是"找最大的"，训练时已经有正确答案了

# 推理/翻译：没有正确答案，用 argmax 取概率最大的词
next_word = torch.argmax(out[:, -1], dim=-1)  # ← 模型自己挑最大的
```

## 八、Warmup 学习率调度

### 是什么
学习率先线性增大到峰值，再缓慢衰减。

### 为什么需要
- 前期：参数随机初始化，大 lr 会让梯度乱跳，小 lr 先找方向
- 后期：需要小步微调才能收敛

### 代码演示
```python
from torch.optim.lr_scheduler import LambdaLR

# 在 Trainer.__init__ 里
warmup_steps = 500

def lr_lambda(step):
    step += 1  # 避免 step=0 时除零
    if step < warmup_steps:
        # 线性增长: 0 → 1（lr 从 0 爬到 base_lr）
        return step / warmup_steps
    else:
        # 衰减: 1 → 0（lr 从 base_lr 慢慢降）
        return warmup_steps ** 0.5 / step ** 0.5

self.scheduler = LambdaLR(self.optimizer, lr_lambda)

# 在 _train_one_step 里调用
self.scaler.step(self.optimizer)
self.scaler.update()
self.scheduler.step()  # ← 每步更新 lr

# 断点续训要保存/恢复 scheduler 状态
# _save_checkpoint:
'scheduler_state_dict': self.scheduler.state_dict()

# _load_checkpoint:
self.scheduler.load_state_dict(checkpoint['scheduler_state_dict'])
```

### lr 变化示例
```python
# warmup_steps=500, base_lr=0.001
# step=1:   lr = 0.001 × (1/500)   = 0.000002
# step=250: lr = 0.001 × (250/500) = 0.0005
# step=500: lr = 0.001 × 1.0       = 0.001  ← 峰值
# step=1000: lr = 0.001 × (500^0.5 / 1000^0.5) = 0.001 × 0.707 = 0.000707
# step=2000: lr = 0.001 × (500^0.5 / 2000^0.5) = 0.001 × 0.5 = 0.0005
```

## 九、评估指标 BLEU

### 是什么
机器翻译标准评估指标，对比机器翻译和参考翻译的 n-gram 重合度。

### 代码演示
```python
from sacrebleu import corpus_bleu

# 生成的翻译
hypotheses = ["hello world", "i love you", "the weather is nice"]
# 参考翻译（正确答案），要包一层 list
references = [["hello world", "i love you", "the weather is good"]]

# 计算 BLEU
bleu = corpus_bleu(hypotheses, references)
print(f"BLEU: {bleu.score}")  # 25.26

# 分数范围:
# < 10: 几乎不可用
# 10~20: 勉强能看
# 20~30: 基本可用  ← 本项目 25.26
# 30~50: 不错
# > 50: 很好
```

### BLEU 怎么算的
```python
# 参考: "the cat is on the mat"
# 生成: "the cat the mat"

# 1-gram (单词匹配):
#   生成: the, cat, the, mat → 4 个词
#   匹配: the ✓, cat ✓, the ✓, mat ✓ → 4 个匹配
#   precision_1 = 4/4 = 1.0

# 2-gram (两词连续匹配):
#   生成: (the cat), (cat the), (the mat) → 3 个
#   匹配: (the cat) ✓, (cat the) ✗, (the mat) ✗ → 1 个匹配
#   precision_2 = 1/3 = 0.33

# BLEU = BP × exp(Σ wn × log(pn))
# BP = Brevity Penalty，惩罚过短的翻译
```

### 评估脚本完整调用
```python
def evaluate_model(model, dataloader, src_tokenizer, tgt_tokenizer, device, loss_fn, config):
    model.eval()
    predictor = Predictor(model, src_tokenizer, tgt_tokenizer, device)
    all_hypotheses, all_references, all_sources = [], [], []
    total_loss, total_tokens = 0, 0

    for batch in tqdm(dataloader, desc="评估"):
        src = batch['src'].to(device)
        tgt = batch['tgt'].to(device)
        src_mask = make_padding_mask(src, pad_id=config.pad_id)

        with torch.no_grad():
            # 逐条贪心解码
            for i in range(src.size(0)):
                output_ids = predictor._greedy_decode(src[i:i+1], src_mask[i:i+1])
                hyp = tgt_tokenizer.decode(output_ids, skip_special_tokens=True)
                ref = tgt_tokenizer.decode(tgt[i].tolist(), skip_special_tokens=True)
                all_hypotheses.append(hyp)
                all_references.append(ref)
                all_sources.append(src_tokenizer.decode(src[i].tolist(), skip_special_tokens=True))

            # 算 batch loss
            batch_data = Batch(src, tgt, pad_id=config.pad_id)
            out = model(batch_data.src, batch_data.tgt, batch_data.src_mask, batch_data.tgt_mask)
            out = model.generator(out)
            loss = loss_fn(out.reshape(-1, out.size(-1)), batch_data.tgt_y.reshape(-1))
            total_loss += loss.item()
            total_tokens += batch_data.ntokens.item()

    # 在循环外算 BLEU + 返回
    bleu = corpus_bleu(all_hypotheses, [all_references])
    results = {"bleu": bleu.score, "loss": total_loss / max(1, total_tokens)}
    return results, all_sources, all_hypotheses, all_references
```

```python
# 1. models.pt（最佳模型权重，推理用）
def _should_early_stop(self, val_loss):
    if val_loss < self.early_stopping_best_score:
        self.early_stopping_best_score = val_loss
        self.early_stopping_counter = 0
        torch.save(self.model.state_dict(), self.best_model_path)  # ← 存最佳模型
        return False
    else:
        self.early_stopping_counter += 1

        'early_stopping_best_score': self.early_stopping_best_score,
        'early_stopping_counter': self.early_stopping_counter,
    }
    torch.save(checkpoint, self.last_ckpt_path)

# 推理时加载 models.pt
model.load_state_dict(torch.load(MODELS_DIR / BEST_MODEL, weights_only=False))

# 续训时加载 checkpoint.pt
checkpoint = torch.load(self.last_ckpt_path, weights_only=False)
self.model.load_state_dict(checkpoint['model_state_dict'])
self.optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
self.scheduler.load_state_dict(checkpoint['scheduler_state_dict'])
self.step = checkpoint['step']
```

### 保存逻辑
```
每次 save_steps 步：
  ├── _should_early_stop() → val_loss 创新低？→ 存 models.pt（最佳模型）
  └── _save_checkpoint() → 存 checkpoint.pt（完整存档）

models.pt 存的是 val_loss 最低时刻的模型，不是最后一步的模型
```

### schemas.py
```python
from pydantic import BaseModel, Field

class PredictRequest(BaseModel):
    text: str = Field(description="请输入中文文本", examples=["你好世界"])

class PredictResponse(BaseModel):
    src: str = Field(description="原始中文")
    tgt: str = Field(description="翻译英文")
```

### routers.py
```python
from fastapi import APIRouter, HTTPException
from web.schemas import PredictRequest, PredictResponse
from web.service import predict_service

predict_router = APIRouter(tags=["翻译接口"])

@predict_router.post("/predict")
def predict(request: PredictRequest) -> PredictResponse:
    try:
        text = request.text.strip()
        if not text:
            raise HTTPException(status_code=400, detail="输入文本不能为空")
        tgt = predict_service(text)
        return PredictResponse(src=text, tgt=tgt)
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"翻译失败: {str(e)}")
```

```python
def make_causal_mask(size, device=None):
```
